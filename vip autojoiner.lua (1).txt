function get_table(p1)
	local function v182(p2)
		local s1 = ""

		for _ = 1, p2 do
			s1 = s1 .. "\t"
		end

		return s1
	end

	local t1 = {}
	local t2 = {}
	local t3 = {}
	local n1 = 1
	local s2 = "{\n"
	local g197 = nil

	while true do
		local n2 = 0

		for _, _ in pairs(p1) do
			n2 = n2 + 1
		end

		local n3 = 1

		for k, v in pairs(p1) do
			if t1[p1] ~= nil and not (n3 >= t1[p1]) then
				if n3 == n2 then
					s2 = s2 .. "\n" .. v182(n1 - 1) .. "}"
				end
			else
				if not string.find(s2, "}", s2:len()) then
					if not string.find(s2, "\n", s2:len()) then
						s2 = s2 .. "\n"
					end
				else
					s2 = s2 .. ",\n"
				end

				table.insert(t3, s2)

				local s3 = ""
				local v195

				if type(k) ~= "number" and type(k) ~= "boolean" then
					v195 = "['" .. tostring(k) .. "']"
				else
					v195 = "[" .. tostring(k) .. "]"
				end

				local v196

				if type(v) ~= "number" and type(v) ~= "boolean" then
					if type(v) == "table" then
						s2 = s3 .. v182(n1) .. v195 .. " = {\n"
						table.insert(t2, p1)
						table.insert(t2, v)
						t1[p1] = n3 + 1

						break
					end

					v196 = s3 .. v182(n1) .. v195 .. " = '" .. tostring(v) .. "'"
				else
					v196 = s3 .. v182(n1) .. v195 .. " = " .. tostring(v)
				end

				if n3 ~= n2 then
					s2 = v196 .. ","
				else
					s2 = v196 .. "\n" .. v182(n1 - 1) .. "}"
				end
			end

			n3 = n3 + 1
		end

		if not (#t2 > 0) then
			break
		end

		p1 = t2[#t2]
		t2[#t2] = nil

		repeat
			if g197 or t1[p1] ~= nil then
				g197 = false
				n1 = n1 - 1
			else
				local v198 = n1 + 1

				if not v198 then
					g197 = true
				end

				if not g197 then
					n1 = v198
				end
			end
		until not g197
	end

	table.insert(t3, s2)

	return (table.concat(t3))
end
